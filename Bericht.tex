% 1. Autoren-Verwaltung
% ===============================

% Leere Liste zum Sammeln der Autoren:
\newcommand{\authorslist}{}

% Einzelnes Autoren-Format:
\newcommand{\authorentry}[3]{%
	\par\noindent
	\textbf{#1}\\
	#2\\
	\texttt{#3}\par\vspace{0.8em}
}

% Befehl zum Hinzufügen eines Autors:
\newcommand{\addauthor}[3]{%
	\g@addto@macro\authorslist{\authorentry{#1}{#2}{#3}}
}

% Befehl zum Ausgeben des Verzeichnisses:
\newcommand{\printauthors}{%
	\section*{Autorenverzeichnis}
	\authorslist
}
